    <script>
        // ===== STATE MANAGEMENT =====
        const state = {
            currentMode: 'idle',
            theme: 'dark',
            focusTime: 25 * 60,
            focusInterval: null,
            wordCount: 0,
            charCount: 0
        };

        const modes = {
            focus: { name: 'Focus Mode', color: '#3b82f6', text: 'Focus Mode Active', loading: 'Preparing deep work environment...' },
            writing: { name: 'Writing Mode', color: '#10b981', text: 'Writing Mode Active', loading: 'Setting up distraction-free editor...' },
            meeting: { name: 'Meeting Mode', color: '#8b5cf6', text: 'Meeting Mode Active', loading: 'Joining meeting room...' },
            study: { name: 'Study Mode', color: '#f59e0b', text: 'Study Mode Active', loading: 'Loading study materials...' },
            break: { name: 'Break Mode', color: '#ec4899', text: 'Break Mode Active', loading: 'Preparing relaxation space...' },
            idle: { name: 'FlowKeys Ready', color: '#5a5a6a', text: 'FlowKeys Ready', loading: 'Resetting workspace...' }
        };

        // ===== DOM ELEMENTS =====
        const modeDot = document.getElementById('modeDot');
        const modeText = document.getElementById('modeText');
        const tooltip = document.getElementById('tooltip');
        const loadingOverlay = document.getElementById('loadingOverlay');
        const loadingText = document.getElementById('loadingText');
        const workspace = document.getElementById('workspace');
        const themeToggle = document.getElementById('themeToggle');
        const ctaOverlay = document.getElementById('ctaOverlay');
        const focusTimer = document.getElementById('focusTimer');
        const writingArea = document.getElementById('writingArea');
        const wordCountEl = document.getElementById('wordCount');
        const charCountEl = document.getElementById('charCount');

        // ===== INITIALIZATION =====
        document.addEventListener('DOMContentLoaded', () => {
            // Show tooltip briefly
            setTimeout(() => tooltip.classList.add('show'), 500);
            setTimeout(() => tooltip.classList.remove('show'), 4000);

            // Show CTA after first interaction
            let ctaShown = false;

            // Bind keys
            document.querySelectorAll('.flow-key').forEach(key => {
                key.addEventListener('click', () => {
                    const mode = key.dataset.mode;
                    switchMode(mode);
                    if (!ctaShown && mode !== 'idle') {
                        setTimeout(() => ctaOverlay.classList.add('show'), 2000);
                        ctaShown = true;
                    }
                });
            });

            // Theme toggle
            themeToggle.addEventListener('click', toggleTheme);

            // Writing stats
            writingArea.addEventListener('input', updateWritingStats);
        });

        // ===== MODE SWITCHING =====
        function switchMode(mode) {
            if (state.currentMode === mode) return;

            const modeConfig = modes[mode];
            const prevMode = state.currentMode;

            // Update active key
            document.querySelectorAll('.flow-key').forEach(k => k.classList.remove('active'));
            document.querySelector(`[data-mode="${mode}"]`)?.classList.add('active');

            // Show loading
            loadingText.textContent = modeConfig.loading;
            loadingOverlay.classList.add('show');

            // Simulate micro-delay
            setTimeout(() => {
                loadingOverlay.classList.remove('show');

                // Hide previous view
                document.querySelectorAll('.mode-view').forEach(v => v.classList.remove('active'));

                // Show new view
                const newView = document.getElementById(`view-${mode}`);
                if (newView) newView.classList.add('active');

                // Update top bar
                modeText.textContent = modeConfig.text;
                modeDot.style.background = modeConfig.color;
                modeDot.classList.add('active');
                setTimeout(() => modeDot.classList.remove('active'), 600);

                // Update background orbs
                updateOrbs(mode);

                // Mode-specific logic
                if (mode === 'focus') startFocusTimer();
                else stopFocusTimer();

                if (mode === 'writing') {
                    writingArea.focus();
                    updateWritingStats();
                }

                state.currentMode = mode;
            }, 300 + Math.random() * 200);
        }

        // ===== FOCUS TIMER =====
        function startFocusTimer() {
            stopFocusTimer();
            state.focusTime = 25 * 60;
            updateTimerDisplay();
            state.focusInterval = setInterval(() => {
                if (state.focusTime > 0) {
                    state.focusTime--;
                    updateTimerDisplay();
                } else {
                    stopFocusTimer();
                    focusTimer.textContent = 'Done!';
                }
            }, 1000);
        }

        function stopFocusTimer() {
            if (state.focusInterval) {
                clearInterval(state.focusInterval);
                state.focusInterval = null;
            }
        }

        function updateTimerDisplay() {
            const mins = Math.floor(state.focusTime / 60).toString().padStart(2, '0');
            const secs = (state.focusTime % 60).toString().padStart(2, '0');
            focusTimer.textContent = `${mins}:${secs}`;
        }

        // ===== WRITING STATS =====
        function updateWritingStats() {
            const text = writingArea.value;
            state.charCount = text.length;
            state.wordCount = text.trim() ? text.trim().split(/\s+/).length : 0;
            wordCountEl.textContent = `${state.wordCount} words`;
            charCountEl.textContent = `${state.charCount} chars`;
        }

        // ===== CHECKLIST =====
        function toggleCheck(item) {
            item.classList.toggle('checked');
            updateProgress();
        }

        function updateProgress() {
            const items = document.querySelectorAll('.checklist-item');
            const checked = document.querySelectorAll('.checklist-item.checked');
            const percent = Math.round((checked.length / items.length) * 100);
            document.getElementById('progressFill').style.width = `${percent}%`;
            document.getElementById('progressText').textContent = `${percent}% completed`;
        }

        // ===== THEME =====
        function toggleTheme() {
            state.theme = state.theme === 'dark' ? 'light' : 'dark';
            document.documentElement.setAttribute('data-theme', state.theme);
            themeToggle.textContent = state.theme === 'dark' ? '🌙' : '☀️';
        }

        // ===== ORBS =====
        function updateOrbs(mode) {
            const orb1 = document.getElementById('orb1');
            const orb2 = document.getElementById('orb2');
            const orb3 = document.getElementById('orb3');

            const colors = {
                focus: ['#3b82f6', '#1d4ed8', '#60a5fa'],
                writing: ['#10b981', '#059669', '#34d399'],
                meeting: ['#8b5cf6', '#7c3aed', '#a78bfa'],
                study: ['#f59e0b', '#d97706', '#fbbf24'],
                break: ['#ec4899', '#db2777', '#f472b6'],
                idle: ['#3b82f6', '#8b5cf6', '#10b981']
            };

            const c = colors[mode] || colors.idle;
            orb1.style.background = `radial-gradient(circle, ${c[0]}, transparent 70%)`;
            orb2.style.background = `radial-gradient(circle, ${c[1]}, transparent 70%)`;
            orb3.style.background = `radial-gradient(circle, ${c[2]}, transparent 70%)`;
        }
    </script>
